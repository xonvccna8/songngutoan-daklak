    // Mặc định từ chối. Quyền theo claims riêng: lsnRole, lsnSchool, lsnClasses.
    // Mọi thao tác ghi lớp/học sinh đi qua API route (firebase-admin) nên client chỉ đọc.
    function lsnSignedIn() { return request.auth != null; }
    function lsnRole() { return request.auth.token.get('lsnRole', ''); }
    function lsnIsAdmin() { return lsnSignedIn() && lsnRole() == 'admin'; }
    function lsnIsTeacher() { return lsnSignedIn() && lsnRole() == 'teacher'; }
    function lsnIsStudent() { return lsnSignedIn() && lsnRole() == 'student'; }
    function lsnSchool() { return request.auth.token.get('lsnSchool', ''); }
    function lsnInMyClasses(classId) { return classId in request.auth.token.get('lsnClasses', []); }
    function lsnClassData(classId) { return get(/databases/$(database)/documents/lsn_classes/$(classId)).data; }
    // GV có tên trong teacherIds, hoặc quản trị cùng trường
    function lsnIsStaffOf(classId) {
      return (lsnIsTeacher() && request.auth.uid in lsnClassData(classId).teacherIds)
          || (lsnIsAdmin() && lsnClassData(classId).schoolId == lsnSchool());
    }

    match /lsn_schools/{schoolId} {
      allow read: if lsnSignedIn() && lsnSchool() == schoolId;
      allow write: if false;
    }

    match /lsn_users/{uid} {
      allow read: if lsnSignedIn() && request.auth.uid == uid;
      allow write: if false;
    }

    match /lsn_classes/{classId} {
      allow read: if (lsnIsTeacher() && request.auth.uid in resource.data.teacherIds)
                  || (lsnIsAdmin() && resource.data.schoolId == lsnSchool())
                  || (lsnIsStudent() && lsnInMyClasses(classId));
      allow write: if false;

      match /students/{uid} {
        // HS chỉ đọc hồ sơ của chính mình (không thấy tên bạn khác qua client)
        allow read: if lsnIsStaffOf(classId)
                    || (lsnIsStudent() && lsnInMyClasses(classId) && request.auth.uid == uid);
        allow write: if false;
      }
    }

    // Hash PIN — chỉ máy chủ (Admin SDK bỏ qua rules)
    match /lsn_private/{document=**} {
      allow read, write: if false;
    }

    // Thống kê: chỉ máy chủ ghi; GV lớp và chính HS được đọc
    match /lsn_studentStats/{id} {
      allow read: if lsnSignedIn() && (
                    lsnIsStaffOf(resource.data.classId)
                    || (lsnIsStudent() && resource.data.uid == request.auth.uid));
      allow write: if false;
    }

    // ===== GĐ 2: tiết học & dạy trực tiếp =====
    function lsnAnyRole() { return lsnIsAdmin() || lsnIsTeacher() || lsnIsStudent(); }
    function lsnSess(sid) { return get(/databases/$(database)/documents/lsn_sessions/$(sid)).data; }

    // Nội dung tiết: GV/quản trị đọc mọi tiết (kể cả nháp); HS chỉ đọc tiết đã duyệt.
    // Đáp án + ghi chú (teacher/*) chỉ GV/quản trị. Ghi tiết chỉ qua API (GĐ 6).
    function lsnIsStaff() { return lsnIsTeacher() || lsnIsAdmin(); }
    match /lsn_lessons/{lessonId} {
      allow read: if lsnIsStaff() || (lsnIsStudent() && resource.data.status == 'approved');
      allow write: if false;
      match /slides/{slideId} {
        allow read: if lsnIsStaff()
                    || (lsnIsStudent() && get(/databases/$(database)/documents/lsn_lessons/$(lessonId)).data.status == 'approved');
        allow write: if false;
      }
      match /teacher/{slideId} {
        allow read: if lsnIsTeacher() || lsnIsAdmin();
        allow write: if false;
      }
    }

    match /lsn_sessions/{sid} {
      allow read: if lsnIsStaffOf(resource.data.classId)
                  || (lsnIsStudent() && lsnInMyClasses(resource.data.classId));
      // GV điều khiển: chỉ được sửa "state" khi tiết đang dạy. Tạo/kết thúc qua API.
      allow update: if lsnIsStaffOf(resource.data.classId)
                    && resource.data.status == 'live'
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['state', 'updatedAt']);
      allow create, delete: if false;

      match /participants/{uid} {
        allow read: if lsnIsStaffOf(lsnSess(sid).classId)
                    || (lsnIsStudent() && request.auth.uid == uid);
        allow create: if lsnIsStudent() && request.auth.uid == uid
                      && lsnInMyClasses(lsnSess(sid).classId) && lsnSess(sid).status == 'live'
                      && request.resource.data.keys().hasOnly(['joinedAt', 'lastSeenAt', 'visible', 'leftCount', 'raisedHandAt'])
                      && request.resource.data.get('leftCount', 0) == 0;
        // HS sửa nhịp/rời trang/giơ tay của mình (không được sửa điểm); GV chỉ được hạ tay.
        // Số lần rời trang chỉ được TĂNG (HS không tự xoá được).
        allow update: if (lsnIsStudent() && request.auth.uid == uid
                          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['lastSeenAt', 'visible', 'leftCount', 'raisedHandAt'])
                          && request.resource.data.get('leftCount', 0) >= resource.data.get('leftCount', 0))
                      || (lsnIsStaffOf(lsnSess(sid).classId)
                          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['raisedHandAt']));
        allow delete: if false;
      }

      // Câu trả lời: chỉ máy chủ ghi (chấm điểm); GV đọc cả lớp; HS đọc của mình.
      // Tiết có cờ hideUntilReveal (trang KHTN): HS chỉ đọc được khi tiết đã kết thúc — lúc đang dạy
      // HS xem phiếu "mine" (đúng/sai chỉ có sau khi GV công bố). Tiết không có cờ (trang Toán cũ): như cũ.
      match /responses/{rid} {
        allow read: if lsnIsStaffOf(lsnSess(sid).classId)
                    || (lsnIsStudent() && resource.data.uid == request.auth.uid
                        && (lsnSess(sid).get('hideUntilReveal', false) != true || lsnSess(sid).status == 'ended'));
        allow write: if false;
      }

      // Phiếu của em: máy chủ ghi (câu đã gửi + kết quả câu đã công bố); HS đọc của mình, GV đọc cả lớp
      match /mine/{uid} {
        allow read: if lsnIsStaffOf(lsnSess(sid).classId)
                    || (lsnIsStudent() && request.auth.uid == uid);
        allow write: if false;
      }
      // qstats (bộ đếm xếp hạng): không có quy tắc → chỉ máy chủ đọc/ghi
    }

    // ===== GĐ 4: tin nhắn & thông báo đẩy =====
    function lsnMsg(mid) { return get(/databases/$(database)/documents/lsn_messages/$(mid)).data; }

    // Tin: chỉ máy chủ tạo (kiểm giới hạn + giờ yên lặng). Đọc: HS được nhận, quản trị cùng trường, GV của lớp
    match /lsn_messages/{mid} {
      allow read: if (lsnIsStudent() && request.auth.uid in resource.data.toUids)
                  || (lsnIsAdmin() && resource.data.schoolId == lsnSchool())
                  || lsnIsStaffOf(resource.data.classId);
      allow write: if false;

      // Biên nhận của từng em: HS chỉ sửa của mình, chỉ các trường trạng thái / trả lời nhanh
      match /receipts/{uid} {
        allow read: if (lsnIsStudent() && request.auth.uid == uid)
                    || (lsnIsAdmin() && lsnMsg(mid).schoolId == lsnSchool())
                    || lsnIsStaffOf(lsnMsg(mid).classId);
        allow update: if lsnIsStudent() && request.auth.uid == uid
                      && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'deliveredAt', 'readAt', 'ackAt', 'reply'])
                      && request.resource.data.status in ['sent', 'delivered', 'read', 'acked']
                      && (request.resource.data.reply == null || request.resource.data.reply in ['understood', 'need_help']);
        allow create, delete: if false;
      }
    }

    // Mã thiết bị nhận thông báo đẩy: mỗi người tự quản lý của mình
    match /lsn_fcmTokens/{uid}/tokens/{tid} {
      allow read, delete: if lsnSignedIn() && request.auth.uid == uid;
      allow create, update: if lsnSignedIn() && request.auth.uid == uid
                            && request.resource.data.keys().hasOnly(['token', 'ua', 'updatedAt'])
                            && request.resource.data.token is string && request.resource.data.token.size() <= 4096;
    }

    // ===== GĐ 5: tự học — mọi thao tác ghi đi qua API (máy chủ chấm điểm, chặn gian lận thời gian) =====
    function lsnHw(aid) { return get(/databases/$(database)/documents/lsn_assignments/$(aid)).data; }

    // Ngân hàng thuật ngữ: GV/quản trị đọc tất cả; HS chỉ đọc thuật ngữ đã duyệt

    match /lsn_events/{eid} {
